~[tlist_sql; WITH school_year AS ( SELECT (EXTRACT(YEAR FROM ADD_MONTHS(SYSDATE, -6)) - 1990) * 100 AS year_term_id FROM dual ) SELECT students.dcid, students.student_number, students.lastfirst, schools.name AS school_name, CASE WHEN students.grade_level < 0 THEN 'PK' WHEN students.grade_level=0 THEN 'K' ELSE TO_CHAR(students.grade_level) END AS grade_label, CASE students.enroll_status WHEN -1 THEN 'Pre-registered' ELSE 'Active' END AS enrollment_record FROM students JOIN schools ON schools.school_number=students.schoolid JOIN u_student_enrollment_logic ON u_student_enrollment_logic.studentsdcid=students.dcid CROSS JOIN school_year WHERE students.schoolid IN (189, 437) AND ~(curschoolid) IN (0, students.schoolid) AND students.enroll_status IN (-1, 0) AND students.grade_level <=12 AND ( u_student_enrollment_logic.preenroll_year_id=school_year.year_term_id OR ( u_student_enrollment_logic.renewal_year_id=school_year.year_term_id AND u_student_enrollment_logic.psee_school_id NOT IN (189, 437) ) ) ORDER BY schools.name, students.grade_level, students.lastfirst, students.id ;nonemessage=] [/tlist_sql]
