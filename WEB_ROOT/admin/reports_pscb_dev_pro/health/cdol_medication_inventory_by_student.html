~[tlist_sql; WITH inventory_totals AS ( SELECT u_student_medication_id, SUM(quantity_added) AS quantity_received FROM u_student_medication_inventory GROUP BY u_student_medication_id ), transaction_totals AS ( SELECT u_student_medication_id, SUM(quantity_change) AS quantity_change FROM u_student_med_inv_txn GROUP BY u_student_medication_id ), student_medications AS ( SELECT medication.studentsdcid, COUNT(medication.id) AS medication_count, '
    ' || LISTAGG( '
  • ' || medication.medication_name || ' - ' || medication.dose_amount || ' ' || NVL(dose_unit_option.displayvalue, medication.dose_unit) || ' | ' || NVL(frequency_option.displayvalue, medication.frequency) || ' | Available ' || GREATEST(0, NVL(inventory_totals.quantity_received, 0) + NVL(transaction_totals.quantity_change, 0)) || ' ' || NVL(inventory_unit_option.displayvalue, medication.inventory_unit) || '
    • ', '' ) WITHIN GROUP (ORDER BY medication.medication_name, medication.dose_amount) || '
' AS medication_list FROM u_student_medication medication LEFT JOIN inventory_totals ON inventory_totals.u_student_medication_id = medication.id LEFT JOIN transaction_totals ON transaction_totals.u_student_medication_id = medication.id LEFT JOIN u_cdol_health_option dose_unit_option ON dose_unit_option.codetype = 'MED_DOSE_UNIT' AND dose_unit_option.code = medication.dose_unit LEFT JOIN u_cdol_health_option inventory_unit_option ON inventory_unit_option.codetype = 'MED_INVENTORY_UNIT' AND inventory_unit_option.code = medication.inventory_unit LEFT JOIN u_cdol_health_option frequency_option ON frequency_option.codetype = 'MED_FREQUENCY' AND frequency_option.code = medication.frequency WHERE medication.yearid = ~(curyearid) ~[if.is.a.school] AND medication.schoolid = ~(curschoolid) [/if.is.a.school] GROUP BY medication.studentsdcid ) SELECT students.dcid AS dcid, students.student_number AS student_number, students.lastfirst AS lastfirst, CASE WHEN UPPER(students.gender) = 'F' THEN 'Female' WHEN UPPER(students.gender) = 'M' THEN 'Male' ELSE students.gender END AS gender, CASE WHEN students.grade_level = -1 THEN 'PK' WHEN students.grade_level = 0 THEN 'K' WHEN students.grade_level = 99 THEN 'Graduated' ELSE TO_CHAR(students.grade_level) END AS grade_level, student_medications.medication_list AS medication_list FROM students LEFT JOIN student_medications ON students.dcid = student_medications.studentsdcid ~[if#ss.~(gpv.ss;sqlText)=cur] INNER JOIN ~[temp.table.current.selection:students] temp ON students.dcid = temp.dcid [/if#ss] WHERE students.enroll_status IN (-1, 0) ~[if.is.a.school] AND students.schoolid = ~(curschoolid) [/if.is.a.school] ~[if#medications.~(gpv.medications;sqlText)=1] AND student_medications.medication_count > 0 [/if#medications] ORDER BY students.lastfirst ;alternatecolor;nonemessage=] [/tlist_sql]
